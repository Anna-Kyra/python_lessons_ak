# EX03: COSMIC DANCE
def draw_star(center_x: float, center_y: float, size: float, color: tuple):
    """
    Draws a 5-point star with a fill color of choice, but no outline. It is drawn as a polygon.
    :param center_x: The center x position of the star
    :param center_y: The center y position of the star
    :param size: The size (diameter) to draw the star with
    :param color: The fill color of the star
    :return: -
    """
    engine.outline_color = None
    engine.color = color
    dots = []
    num_points = 5
    outer_radius = size / 2
    inner_radius = outer_radius / 2
    angle = math.pi / num_points

    for i in range(num_points * 2):
        r = outer_radius if i % 2 == 0 else inner_radius
        x = center_x + math.cos(i * angle) * r
        y = center_y + math.sin(i * angle) * r
        dots.append((x, y))

    engine.draw_poly(dots)
